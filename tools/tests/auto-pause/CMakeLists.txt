cmake_minimum_required(VERSION 3.25)
project(auto_pause_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(rexglue 0.10.0 REQUIRED CONFIG)
enable_testing()
set(test_sources auto_pause_test controller_disconnect_probe)
if(WIN32)
    list(APPEND test_sources auto_pause_native_probe)
endif()
foreach(name IN LISTS test_sources)
    add_executable(${name} ../../../forza-recomp/tests/${name}.cpp)
    target_link_libraries(${name} PRIVATE rex::runtime)
    if(MSVC)
        target_compile_options(${name} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${name} PRIVATE -UNDEBUG)
    endif()
    if(WIN32)
        target_compile_definitions(${name} PRIVATE NOMINMAX)
        add_custom_command(TARGET ${name} POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "$<TARGET_FILE:rex::runtime>" "$<TARGET_FILE_DIR:${name}>")
    endif()
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 30)
endforeach()
set_tests_properties(controller_disconnect_probe PROPERTIES
    PASS_REGULAR_EXPRESSION "SDK disconnect probe: active loss")
if(WIN32)
    set_tests_properties(auto_pause_native_probe PROPERTIES
        PASS_REGULAR_EXPRESSION "native auto-pause: live paused")
endif()
