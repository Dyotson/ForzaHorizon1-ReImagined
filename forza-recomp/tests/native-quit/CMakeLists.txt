cmake_minimum_required(VERSION 3.25)
project(native_quit_adapter_checks LANGUAGES CXX)
find_package(rexglue 0.10.0 REQUIRED CONFIG)
add_executable(native_quit_test ../native_quit_test.cpp)
target_compile_features(native_quit_test PRIVATE cxx_std_23)
target_include_directories(native_quit_test PRIVATE ../../src)
target_compile_definitions(native_quit_test PRIVATE FORZA_TITLE_UPDATE=4 FORZA_NATIVE_PAUSE=1)
target_link_libraries(native_quit_test PRIVATE rex::runtime)
if(MSVC)
    target_compile_options(native_quit_test PRIVATE /UNDEBUG)
else()
    target_compile_options(native_quit_test PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME native_quit_adapter COMMAND native_quit_test)
