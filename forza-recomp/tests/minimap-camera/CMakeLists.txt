cmake_minimum_required(VERSION 3.25)
project(minimap_camera_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(rexglue 0.10.0 REQUIRED CONFIG)
find_package(Threads REQUIRED)
add_executable(minimap_heading_test ../minimap_heading_test.cpp)
target_link_libraries(minimap_heading_test PRIVATE Threads::Threads)
add_executable(minimap_camera_test ../minimap_camera_test.cpp)
target_compile_definitions(minimap_camera_test PRIVATE FORZA_TITLE_UPDATE=4)
target_include_directories(minimap_camera_test PRIVATE ../../src)
target_link_libraries(minimap_camera_test PRIVATE rex::runtime)
enable_testing()
add_test(NAME minimap_heading COMMAND minimap_heading_test)
add_test(NAME minimap_camera_adapter COMMAND minimap_camera_test)
