cmake_minimum_required(VERSION 3.25)
project(horizon2_ui_navigation_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(REXSDK_INCLUDE_DIR "" CACHE PATH "ReXGlue SDK include directory")
set(FH2_NAVIGATION_DIR "" CACHE PATH "Private directory containing pause_navigation.inc")
if(NOT EXISTS "${REXSDK_INCLUDE_DIR}/rex/ppc/context.h" OR
   NOT EXISTS "${FH2_NAVIGATION_DIR}/pause_navigation.inc")
    message(FATAL_ERROR "Set REXSDK_INCLUDE_DIR and FH2_NAVIGATION_DIR; see docs/horizon2-menu-port.md")
endif()
add_executable(horizon2_navigation_test ../horizon2_navigation_test.cpp)
target_include_directories(horizon2_navigation_test PRIVATE "${REXSDK_INCLUDE_DIR}" "${FH2_NAVIGATION_DIR}")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" AND CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|x86_64")
    target_compile_options(horizon2_navigation_test PRIVATE -msse4.1)
endif()
if(MSVC)
    target_compile_options(horizon2_navigation_test PRIVATE /UNDEBUG)
else()
    target_compile_options(horizon2_navigation_test PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME original_horizon2_navigation COMMAND horizon2_navigation_test)
