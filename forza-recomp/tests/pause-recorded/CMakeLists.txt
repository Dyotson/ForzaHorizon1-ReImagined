cmake_minimum_required(VERSION 3.25)
project(pause_recorded_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(pause_fixed_pose_hold ../pause_fixed_pose_hold_test.cpp)
add_executable(pause_recorded_owner ../pause_recorded_owner_test.cpp)
add_executable(pause_recorded_tu0 ../pause_recorded_stubs_test.cpp)
target_compile_definitions(pause_recorded_tu0 PRIVATE FORZA_TITLE_UPDATE=0 FORZA_NATIVE_PAUSE=1)
add_executable(pause_recorded_disabled ../pause_recorded_stubs_test.cpp)
target_compile_definitions(pause_recorded_disabled PRIVATE FORZA_TITLE_UPDATE=4)
foreach(target IN ITEMS pause_fixed_pose_hold pause_recorded_owner pause_recorded_tu0 pause_recorded_disabled)
    if(MSVC)
        target_compile_options(${target} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${target} COMMAND ${target})
endforeach()
