cmake_minimum_required(VERSION 3.25)
project(forza-discord-tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(FORZA_DISCORD_TEST_TRANSPORT ON)
add_executable(forza ../discord_probe.cpp)
# Keep assertions enabled in Release builds too.
if(MSVC)
    target_compile_options(forza PRIVATE /UNDEBUG)
else()
    target_compile_options(forza PRIVATE -UNDEBUG)
endif()
include(../../cmake/discord.cmake)
enable_testing()
add_test(NAME discord-model COMMAND forza)
