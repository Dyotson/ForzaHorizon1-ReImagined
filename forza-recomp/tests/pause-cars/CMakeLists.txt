cmake_minimum_required(VERSION 3.25)
project(pause_cars_checks LANGUAGES CXX)
enable_testing()
add_executable(pause_cars_test ../pause_cars_test.cpp ../../src/hooks/offline_cars.cpp)
target_include_directories(pause_cars_test PRIVATE stubs ../../src)
target_compile_features(pause_cars_test PRIVATE cxx_std_20)
target_compile_definitions(pause_cars_test PRIVATE FORZA_TITLE_UPDATE=4 FORZA_NATIVE_PAUSE=1)
if(MSVC)
  target_compile_options(pause_cars_test PRIVATE /W4 /WX /UNDEBUG)
else()
  target_compile_options(pause_cars_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME pause_cars_access_and_reload COMMAND pause_cars_test)
