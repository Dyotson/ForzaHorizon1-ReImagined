cmake_minimum_required(VERSION 3.25)
project(horizon2_grid_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(REXSDK_INCLUDE_DIR "" CACHE PATH "ReXGlue SDK include directory")
set(FH2_GRID_EVENTS_DIR "" CACHE PATH "Private directory containing grid_events.inc")
if(NOT EXISTS "${REXSDK_INCLUDE_DIR}/rex/ppc/context.h" OR
   NOT EXISTS "${FH2_GRID_EVENTS_DIR}/grid_events.inc")
  message(FATAL_ERROR "Set REXSDK_INCLUDE_DIR and FH2_GRID_EVENTS_DIR")
endif()
add_executable(horizon2_grid_events_test ../horizon2_grid_events_test.cpp)
target_include_directories(horizon2_grid_events_test PRIVATE "${REXSDK_INCLUDE_DIR}" "${FH2_GRID_EVENTS_DIR}")
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" AND CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64")
  target_compile_options(horizon2_grid_events_test PRIVATE -msse4.1)
endif()
if(MSVC)
  target_compile_options(horizon2_grid_events_test PRIVATE /UNDEBUG)
else()
  target_compile_options(horizon2_grid_events_test PRIVATE -UNDEBUG)
endif()
enable_testing()
add_test(NAME original_horizon2_grid_events COMMAND horizon2_grid_events_test)

add_executable(horizon2_tab_layout_test ../horizon2_tab_layout_test.cpp)
target_include_directories(horizon2_tab_layout_test PRIVATE "${REXSDK_INCLUDE_DIR}")
if(MSVC)
  target_compile_options(horizon2_tab_layout_test PRIVATE /UNDEBUG)
else()
  target_compile_options(horizon2_tab_layout_test PRIVATE -UNDEBUG)
endif()
add_test(NAME native_horizon2_tab_layout COMMAND horizon2_tab_layout_test)

add_executable(horizon2_cars_layout_test ../horizon2_cars_layout_test.cpp)
target_include_directories(horizon2_cars_layout_test PRIVATE "${REXSDK_INCLUDE_DIR}")
if(MSVC)
  target_compile_options(horizon2_cars_layout_test PRIVATE /UNDEBUG)
else()
  target_compile_options(horizon2_cars_layout_test PRIVATE -UNDEBUG)
endif()
add_test(NAME native_horizon2_cars_layout COMMAND horizon2_cars_layout_test)
